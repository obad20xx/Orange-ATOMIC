{{- if eq $svcType "LoadBalancer" }}

Getting the application URL (LoadBalancer):
  It may take 1-2 minutes for the LoadBalancer IP to be provisioned.

  export APP_IP=$(kubectl get svc {{ include "tasky.releaseName" . }}-frontend \
    -n {{ .Release.Namespace }} \
    -o jsonpath='{.status.loadBalancer.ingress[0].ip}')

  echo "App:        http://$APP_IP"
  echo "Swagger UI: http://$APP_IP/swagger-ui.html"
  echo "API Docs:   http://$APP_IP/v3/api-docs"
  echo "Health:     http://$APP_IP/actuator/health"

{{- else if eq $svcType "NodePort" }}

Getting the application URL (NodePort):
  export NODE_IP=$(kubectl get nodes -n {{ .Release.Namespace }} \
    -o jsonpath='{.items[0].status.addresses[0].address}')
  export NODE_PORT=$(kubectl get svc {{ include "tasky.releaseName" . }}-frontend \
    -n {{ .Release.Namespace }} \
    -o jsonpath='{.spec.ports[0].nodePort}')
  echo "App: http://$NODE_IP:$NODE_PORT"

{{- else }}

Getting the application URL (ClusterIP / port-forward):
  kubectl port-forward svc/{{ include "tasky.releaseName" . }}-frontend 8080:80 \
    -n {{ .Release.Namespace }}

  Then open: http://localhost:8080

{{- end }}

Checking pod status:
  kubectl get pods -n {{ .Release.Namespace }} -l app={{ include "tasky.releaseName" . }}-backend
  kubectl get pods -n {{ .Release.Namespace }} -l app={{ include "tasky.releaseName" . }}-frontend

Tailing backend logs:
  kubectl logs -f deployment/{{ include "tasky.releaseName" . }}-backend \
    -n {{ .Release.Namespace }}
